<div class="container">
    <h2>Quản lý Cookie</h2>

    <div id="loginForm" class="hidden">
        <label>Nhập tên của bạn:</label>
        <input type="text" id="usernameInput" placeholder="Ví dụ: Nguyen Van A">
        <button onclick="handleSave()">Lưu</button>
    </div>

    <div id="welcomePanel" class="hidden">
        <h3 id="greeting"></h3>
        <hr>
        <label>Cập nhật tên mới:</label>
        <input type="text" id="updateInput" placeholder="Tên mới...">
        <button onclick="handleUpdate()">Cập nhật</button>
        <br><br>
        <button class="btn-delete" onclick="handleDelete()">Xóa thông tin</button>
    </div>

<script>
    // --- 1. HÀM HỖ TRỢ XỬ LÝ COOKIE ---

    // Hàm tạo Cookie (Dùng cho cả Tạo mới và Cập nhật)
    function setCookie(cname, cvalue, exdays) {
        const d = new Date();
        // Tính toán thời gian hết hạn: Thời gian hiện tại + số ngày (exdays)
        d.setTime(d.getTime() + (exdays * 24 * 60 * 60 * 1000));
        let expires = "expires="+ d.toUTCString();
        // Cú pháp chuẩn: tên=giá_trị; thời_gian; đường_dẫn
        document.cookie = cname + "=" + cvalue + ";" + expires + ";path=/";
    }

    // Hàm lấy giá trị Cookie theo tên
    function getCookie(cname) {
        let name = cname + "=";
        let decodedCookie = decodeURIComponent(document.cookie);
        let ca = decodedCookie.split(';');
        for(let i = 0; i < ca.length; i++) {
            let c = ca[i];
            while (c.charAt(0) == ' ') {
                c = c.substring(1);
            }
            if (c.indexOf(name) == 0) {
                return c.substring(name.length, c.length);
            }
        }
        return "";
    }

    // Hàm xóa Cookie
    function deleteCookie(cname) {
        // Đặt thời gian hết hạn về quá khứ để trình duyệt tự xóa
        document.cookie = cname + "=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;";
    }

    // --- 2. HÀM XỬ LÝ GIAO DIỆN (LOGIC BÀI TẬP) ---

    // Kiểm tra Cookie khi tải trang
    function checkCookie() {
        let user = getCookie("username");
        const loginForm = document.getElementById("loginForm");
        const welcomePanel = document.getElementById("welcomePanel");
        const greeting = document.getElementById("greeting");

        if (user != "") {
            // Nếu CÓ cookie: Ẩn form nhập, hiện form chào
            loginForm.classList.add("hidden");
            welcomePanel.classList.remove("hidden");
            greeting.innerText = "Xin chào, " + user;
        } else {
            // Nếu KHÔNG có cookie: Hiện form nhập, ẩn form chào
            loginForm.classList.remove("hidden");
            welcomePanel.classList.add("hidden");
        }
    }

    // Xử lý nút Lưu
    function handleSave() {
        const name = document.getElementById("usernameInput").value;
        if (name) {
            setCookie("username", name, 7); // Lưu 7 ngày
            alert("Đã lưu Cookie!");
            checkCookie(); // Tải lại giao diện
        } else {
            alert("Vui lòng nhập tên!");
        }
    }

    // Xử lý nút Cập nhật
    function handleUpdate() {
        const newName = document.getElementById("updateInput").value;
        if (newName) {
            setCookie("username", newName, 7); // Ghi đè cookie cũ
            alert("Đã cập nhật Cookie!");
            // Xóa input cập nhật cho đẹp
            document.getElementById("updateInput").value = ""; 
            checkCookie(); // Tải lại giao diện để hiện tên mới
        } else {
            alert("Vui lòng nhập tên mới!");
        }
    }

    // Xử lý nút Xóa
    function handleDelete() {
        deleteCookie("username");
        alert("Đã xóa thông tin!");
        checkCookie(); // Tải lại giao diện (quay về form nhập)
    }

    // --- 3. KHỞI CHẠY ---
    // Chạy hàm kiểm tra ngay khi tải trang
    window.onload = checkCookie;




//nghia dep trai
</script>
